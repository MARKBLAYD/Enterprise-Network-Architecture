enable
configure terminal

! Создаем VLANы
vlan 10
 name PERM_VLAN10
exit
vlan 20
 name PERM_VLAN20
exit

! Настройка порта для PC4 в access режиме для VLAN 10
interface fastEthernet 0/4
 switchport mode access
 switchport access vlan 10
 no shutdown
 exit

! Настройка транк порта до HQ-BORDER - ДОБАВЛЯЕМ ИНКАПСУЛЯЦИЮ
interface gigabitEthernet 0/1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20
 no shutdown
 exit

! Настройка транк порта до PERM-SW-2 - ДОБАВЛЯЕМ ИНКАПСУЛЯЦИЮ
interface gigabitEthernet 0/2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20
 no shutdown
 exit

end
write memory
