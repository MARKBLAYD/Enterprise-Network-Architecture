enable
configure terminal

! Создаем VLANы
vlan 10
 name PERM_VLAN10
exit
vlan 20
 name PERM_VLAN20
exit

! Настройка портов для PC5 и PC6 в access режиме для VLAN 20
interface range fastEthernet 0/5-6
 switchport mode access
 switchport access vlan 20
 no shutdown
 exit

! Настройка транк порта до PERM-SW-1 - ДОБАВЛЯЕМ ИНКАПСУЛЯЦИЮ
interface gigabitEthernet 0/2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20
 no shutdown
 exit

end
write memory
