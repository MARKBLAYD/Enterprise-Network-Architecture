enable
configure terminal

! Настройка интерфейса в сторону интернета (к 8.8.8.8)
interface gigabitEthernet 0/1
 ip address 8.8.8.1 255.255.255.0
 no shutdown
 exit

! Настройка интерфейса в сторону клиента (HQ-BORDER)!
interface gigabitEthernet 0/0
 ip address 200.200.200.1 255.255.255.252
 no shutdown
 exit

! Настройка статического маршрута "в интернет"
ip route 0.0.0.0 0.0.0.0 8.8.8.8

! Настройка маршрута до сетей клиента
ip route 10.10.0.0 255.255.0.0 200.200.200.2
ip route 10.20.0.0 255.255.0.0 200.200.200.2

end
write memory
