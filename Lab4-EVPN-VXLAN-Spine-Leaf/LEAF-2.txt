/system identity
set name=LEAF-2

/interface bridge
add name=lo0

/ip address
add address=10.200.0.2/32 interface=lo0
add address=10.200.1.6/30 interface=ether1 network=10.200.1.4

/routing ospf instance
add disabled=no name=default router-id=10.200.0.2

/routing ospf area
add disabled=no instance=default name=backbone

/routing ospf interface-template
add area=backbone disabled=no networks=10.200.0.0/24
add area=backbone disabled=no networks=10.200.1.0/24
/routing bgp instance
add as=65000 name=default router-id=10.200.0.2

/routing bgp connection
add afi=evpn instance=default local.address=10.200.0.2 \
    local.role=ibgp multihop=yes name=to-spine-101 \
    remote.address=10.200.0.101 remote.as=65000
/interface bridge
add name=NVE vlan-filtering=yes

/interface vxlan
add bridge=NVE learning=no local-address=10.200.0.2 \
    name=vni1010_vlan10 vni=1010 bridge-pvid=10
    
add bridge=NVE learning=no local-address=10.200.0.2 \
    name=vni1020_vlan20 vni=1020 bridge-pvid=20
/routing bgp evpn
add disabled=no instance=default name=vni1010_vlan10 vni=1010
    
add disabled=no instance=default name=vni1020_vlan20 vni=1020
/interface bridge port
add bridge=NVE frame-types=admit-only-untagged-and-priority-tagged \
    interface=ether2 pvid=10  # VPC-21 в VLAN 10
    
add bridge=NVE frame-types=admit-only-untagged-and-priority-tagged \
    interface=ether3 pvid=20  # VPC-22 в VLAN 20
